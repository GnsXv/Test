local function New(cls, props)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do o[k] = v end
    return o
end

local function Corner(r, p)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r)
    c.Parent = p
    return c
end

local function Stroke(col, th, p, tr)
    local s = Instance.new("UIStroke")
    s.Color = col
    s.Thickness = th or 1.2
    s.Transparency = tr or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = p
    return s
end

local function Tw(obj, props, t, style)
    TweenService:Create(obj,
        TweenInfo.new(t or 0.2, style or Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        props):Play()
end

-- DRAG — so no header
-- ──────────────────────────────
do
    local dragging, dragStart, startPos
    header.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = i.Position
            startPos  = main.Position
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement
        or i.UserInputType == Enum.UserInputType.Touch then
            local d = i.Position - dragStart
            main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local stealBarLoaded = false

local function openStealBar()
    -- se já carregou, só mostra
    local existing = CoreGui:FindFirstChild("VexonStealHUD")
    if existing then
        existing.Enabled = true
        return
    end
    -- primeira vez: executa o VexonStealBar
    -- substitua o conteudo abaixo pelo seu loadstring/require do VexonStealBar
    if getgenv and getgenv()._VexonStealBarLoad then
        pcall(getgenv()._VexonStealBarLoad)
    end
end

local function closeStealBar()
    local existing = CoreGui:FindFirstChild("VexonStealHUD")
    if existing then
        existing.Enabled = false
    end
    -- desativa o steal também
    local cfg = getStealCfg()
    if cfg then cfg.AutoStealEnabled = false end
end

-- TROCA DE TABS
-- ──────────────────────────────
local activeTab = nil

local function switchTab(name)
    if activeTab == name then return end
    activeTab = name

    for n, t in pairs(tabBtns) do
        local active = (n == name)
        tabActive[n] = active
        Tw(t.btn, {
            BackgroundColor3       = active and C.TabActive or C.TabBg,
            BackgroundTransparency = active and 0 or 0.2,
        }, 0.2)
        Tw(t.btn, {TextColor3 = active and C.White or C.TextDim}, 0.2)
        Tw(t.ind, {BackgroundTransparency = active and 0.1 or 1.0}, 0.2)
    end

    clearScroll()
    local build = tabContent[name]
    if build then build() end
end

for name, t in pairs(tabBtns) do
    t.btn.MouseButton1Click:Connect(function() switchTab(name) end)
end

switchTab("STEAL")
